import { test, expect } from '@playwright/test';

test.describe('Benchmark: demobank Transfer & Balance Assertion', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/apps/demobank');
    await page.getByTestId('auth-pin-input').fill('8842');
    await page.getByTestId('auth-submit-btn').click();
  });

  test('should atomically decrement ledger on instant transfer', async ({ page }) => {
    await page.getByTestId('nav-transfer-link').click();
    await page.getByTestId('recipient-account-select').selectOption('acct_savings_09');
    await page.getByTestId('account-transfer-amount').fill('1250.00');
    await page.getByTestId('btn-authorize-wire').click();

    await expect(page.getByTestId('transfer-success-toast')).toBeVisible();
    await expect(page.getByTestId('balance-total-display')).toHaveText('$24,190.50');
  });
});
